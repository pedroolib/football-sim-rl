**Tech Stack:** Python 3.11+ (env 3.14), NumPy, Gymnasium, Stable-Baselines3 + PyTorch, pytest, Ruff.

**Spec:** `docs/superpowers/specs/2026-10-02-m7-drive-qb-design.md` (approved 2026-10-02).

```python
class RouteName(StrEnum):
    """Initial routes available to the receiver."""

    GO = "go"
    SLANT = "slant"
    POST = "post"
    OUT = "out"
    CURL = "curl"
    FLAT = "flat"
    WHEEL = "wheel"
    CORNER = "corner"


ROUTE_OBSERVATION_ORDER: tuple[RouteName, ...] = (
    RouteName.GO,
    RouteName.SLANT,
    RouteName.POST,
    RouteName.OUT,
    RouteName.CURL,
    RouteName.FLAT,
    RouteName.WHEEL,
    RouteName.CORNER,
)
DEFAULT_ROUTE_POOL = ROUTE_OBSERVATION_ORDER


def sample_routes(
    rng: np.random.Generator,
    count: int,
    pool: tuple[RouteName, ...] = DEFAULT_ROUTE_POOL,
) -> tuple[RouteName, ...]:
    """Draw ``count`` routes uniformly from ``pool`` using ``rng``."""
    if count < 0:
        raise ValueError("count must be non-negative")
    if not pool:
        raise ValueError("pool must not be empty")
    indices = rng.integers(0, len(pool), size=count)
    return tuple(pool[int(index)] for index in indices)
```

```python
def _angled_cut(
    start: Vector2,
    sign: float,
    angle_degrees: float,
    length_yards: float,
    field: FieldBounds,
) -> Vector2:
    angle = radians(angle_degrees)
    offset = vec2(length_yards * cos(angle), sign * length_yards * sin(angle))
    return _clamp_to_field(start + offset, field)
```

```python
    def __init__(
        self,
        config: SimulationConfig | None = None,
        seed: int = 0,
        line_of_scrimmage: float | None = None,
        goal_line: float | None = None,
    ) -> None:
        self.config = config if config is not None else SimulationConfig()
        self.seed = int(seed)
        self._rng = np.random.default_rng(self.seed)
        self._line_of_scrimmage = (
            float(line_of_scrimmage)
            if line_of_scrimmage is not None
            else float(self.config.quarterback_start[0])
        )
        self._shift = self._line_of_scrimmage - float(self.config.quarterback_start[0])
        if goal_line is not None and not 0.0 < float(goal_line) < self.config.field.length_yards:
            raise ValueError("goal_line must be inside the field")
        self._goal_line = None if goal_line is None else float(goal_line)
        self._validate_formation_shift()
        self._receiver_ids: list[str] = []
        self._defender_ids: list[str] = []
        self._rusher_ids: list[str] = []
        self.state: PlayState
        self.reset()

    @property
    def line_of_scrimmage(self) -> float:
        """Effective line of scrimmage, in yards."""
        return self._line_of_scrimmage

    def _validate_formation_shift(self) -> None:
        if self._shift == 0.0:
            return
        starts = (
            (self.config.quarterback_start,)
            + tuple(self.config.receiver_starts)
            + tuple(self.config.defender_starts)
            + tuple(self.config.rusher_starts)
        )
        for start in starts:
            if not self.config.field.contains(vec2(start[0] + self._shift, start[1])):
                raise ValueError("line_of_scrimmage shifts players outside the field")
```

```python
"""Down-and-distance drive rules layered on top of single plays."""

from dataclasses import dataclass

import numpy as np

from qb_sim.config import SimulationConfig
from qb_sim.engine.entities import PlayOutcome
from qb_sim.rules.play import Play
from qb_sim.rules.routes import DEFAULT_ROUTE_POOL, RouteName, sample_routes


@dataclass(frozen=True, slots=True)
class DriveConfig:
    """Downs, field-position range, and arcade scoring for a drive."""

    downs: int = 3
    first_down_yards: float = 10.0
    los_range: tuple[float, float] = (25.0, 50.0)
    max_plays: int = 20
    touchdown_points: float = 7.0
    goal_line: float = 100.0

    def __post_init__(self) -> None:
        if self.downs < 1:
            raise ValueError("downs must be at least 1")
        if self.first_down_yards <= 0.0:
            raise ValueError("first_down_yards must be positive")
        low, high = self.los_range
        if not 0.0 <= low <= high < self.goal_line:
            raise ValueError("los_range must be inside the field")
        if self.max_plays < 1:
            raise ValueError("max_plays must be at least 1")
        if self.touchdown_points <= 0.0:
            raise ValueError("touchdown_points must be positive")


@dataclass(slots=True)
class DriveState:
    """Mutable down, distance, and field-position state for one drive."""

    los: float
    down: int = 1
    distance: float = 10.0
    play_number: int = 0
    first_downs: int = 0
    points: float = 0.0
    terminated: bool = False
    end_reason: str | None = None


@dataclass(frozen=True, slots=True)
class PlayResult:
    """Effective result of one play, already interpreted by the drive rules."""

    outcome: PlayOutcome
    spot: float
    yards: float
    first_down: bool
    touchdown: bool
    drive_over: bool
    end_reason: str | None
    points: float
    throw_distance: float | None
    coverage_sack: bool = False


class Drive:
    """Deterministic 3-down drive: LOS, routes, and play outcomes."""

    def __init__(
        self,
        config: SimulationConfig | None = None,
        drive_config: DriveConfig | None = None,
        seed: int = 0,
    ) -> None:
        self.config = config if config is not None else SimulationConfig()
        self.drive_config = drive_config if drive_config is not None else DriveConfig()
        self.seed = int(seed)
        self._rng = np.random.default_rng(self.seed)
        self._routes: tuple[RouteName, ...] = ()
        self.state: DriveState
        self.reset()

    @property
    def routes(self) -> tuple[RouteName, ...]:
        """Routes sampled for the current play."""
        return self._routes

    def reset(self, seed: int | None = None) -> DriveState:
        """Start a new drive, sampling a fresh line of scrimmage."""
        if seed is not None:
            self.seed = int(seed)
            self._rng = np.random.default_rng(self.seed)
        low, high = self.drive_config.los_range
        los = float(self._rng.uniform(low, high))
        distance = min(self.drive_config.first_down_yards, self.drive_config.goal_line - los)
        self.state = DriveState(los=los, distance=distance)
        self._routes = ()
        return self.state

    def start_play(self) -> Play:
        """Sample routes and return the next mechanical play at the current LOS."""
        if self.state.terminated:
            raise ValueError("the drive is already over")
        self.state.play_number += 1
        self._routes = sample_routes(
            self._rng,
            len(self.config.receiver_starts),
            DEFAULT_ROUTE_POOL,
        )
        play_seed = int(self._rng.integers(0, 2**31 - 1))
        return Play(
            self.config,
            seed=play_seed,
            line_of_scrimmage=self.state.los,
            goal_line=self.drive_config.goal_line,
        )

    def advance(self, play: Play) -> PlayResult:
        """Apply a finished play to the drive state and return its result."""
        state = play.state
        old_los = self.state.los
        outcome = state.outcome
        coverage_sack = False
        if outcome is PlayOutcome.TIME_LIMIT:
            if state.throw_time is None and state.carrier_id is None:
                outcome = PlayOutcome.SACK
                coverage_sack = True
            else:
                outcome = PlayOutcome.INCOMPLETE
        spot = self._spot(state, outcome, old_los)
        yards = spot - old_los
        touchdown = outcome is PlayOutcome.TOUCHDOWN
        first_down = False
        points = 0.0
        end_reason: str | None = None

        if touchdown:
            points = self.drive_config.touchdown_points
            self.state.points += points
            self.state.terminated = True
            end_reason = "touchdown"
        elif outcome is PlayOutcome.INTERCEPTION:
            self.state.terminated = True
            end_reason = "interception"
        elif spot - old_los >= self.state.distance:
            first_down = True
            self.state.first_downs += 1
            self.state.down = 1
            self.state.los = spot
            self.state.distance = min(
                self.drive_config.first_down_yards,
                self.drive_config.goal_line - spot,
            )
        else:
            marker = old_los + self.state.distance
            self.state.los = spot
            self.state.down += 1
            self.state.distance = marker - spot
            if self.state.down > self.drive_config.downs:
                self.state.terminated = True
                end_reason = "turnover_on_downs"

        if not self.state.terminated and self.state.play_number >= self.drive_config.max_plays:
            self.state.terminated = True
            end_reason = "max_plays"

        self.state.end_reason = end_reason

        throw_distance = (
            float(state.ball.flight_distance) if state.throw_time is not None else None
        )
        return PlayResult(
            outcome=outcome,
            spot=spot,
            yards=yards,
            first_down=first_down,
            touchdown=touchdown,
            drive_over=self.state.terminated,
            end_reason=end_reason,
            points=points,
            throw_distance=throw_distance,
            coverage_sack=coverage_sack,
        )

    def _spot(self, state, outcome: PlayOutcome, los: float) -> float:
        goal_line = self.drive_config.goal_line
        if outcome is PlayOutcome.TOUCHDOWN:
            return goal_line
        if outcome is PlayOutcome.COMPLETED and state.carrier_id is not None:
            carrier_x = float(state.player(state.carrier_id).position[0])
            return float(min(max(carrier_x, 0.0), goal_line))
        if outcome is PlayOutcome.SACK:
            quarterback_x = float(state.player("qb").position[0])
            return float(min(max(quarterback_x, 0.0), goal_line))
        return float(los)
```

```python
"""Fixed-size drive observation for the quarterback environment (v11).

Layout (57 floats in ``[-1, 1]``): the 36 passing-environment values first,
then route one-hots per receiver slot (8 routes each), then down one-hot (3),
distance to the marker normalized by 40, and yards to the goal line.
"""

import numpy as np

from qb_sim.config import SimulationConfig
from qb_sim.engine.entities import PlayState
from qb_sim.rules.drive import DriveState
from qb_sim.rules.routes import ROUTE_OBSERVATION_ORDER, RouteName
from qb_sim.training.observations import OBSERVATION_SIZE, observation_from_state

RECEIVER_SLOTS = 2
ROUTE_SLOTS = len(ROUTE_OBSERVATION_ORDER)
DOWN_SLOTS = 3
DISTANCE_SCALE = 40.0
DRIVE_OBSERVATION_SIZE = OBSERVATION_SIZE + RECEIVER_SLOTS * ROUTE_SLOTS + DOWN_SLOTS + 2


def drive_observation_from_state(
    state: PlayState,
    routes: tuple[RouteName, ...],
    drive_state: DriveState,
    config: SimulationConfig,
    goal_line: float = 100.0,
) -> np.ndarray:
    """Extend the QB observation with route, down, and field-position context."""
    values = [float(value) for value in observation_from_state(state, config)]
    for slot in range(RECEIVER_SLOTS):
        route = routes[slot] if slot < len(routes) else None
        values.extend(
            1.0 if route is route_name else 0.0 for route_name in ROUTE_OBSERVATION_ORDER
        )
    down = int(drive_state.down)
    values.extend(1.0 if down == index else 0.0 for index in range(1, DOWN_SLOTS + 1))
    values.append(min(float(drive_state.distance), DISTANCE_SCALE) / DISTANCE_SCALE)
    values.append((float(goal_line) - float(drive_state.los)) / float(goal_line))
    return np.clip(np.array(values, dtype=np.float32), -1.0, 1.0)
```

```python
"""Gymnasium environment whose episode is a 3-down drive (M7).

The quarterback is the learning agent; receivers, corners, and rushers use the
M3 rule controllers with the drive's sampled routes. The observation is the
57-float drive layout and the action is the same 6-value hybrid action.
"""

from collections.abc import Mapping
from typing import Any

import gymnasium as gym
import numpy as np
from gymnasium import spaces

from qb_sim.agents.base import BaseAgent
from qb_sim.agents.loop import build_rule_agents
from qb_sim.engine.actions import Move, Throw
from qb_sim.engine.entities import PlayOutcome
from qb_sim.rules.drive import Drive, DriveConfig
from qb_sim.rules.play import Play
from qb_sim.rules.scenarios import RUSHED_SCENARIO, Scenario
from qb_sim.training.actions import ACTION_SIZE, decode_action
from qb_sim.training.drive_observations import (
    DRIVE_OBSERVATION_SIZE,
    drive_observation_from_state,
)
from qb_sim.training.rewards import DriveRewardConfig, drive_result_reward


class QBDriveEnv(gym.Env[np.ndarray, np.ndarray]):
    """Train a quarterback to move the chains and score touchdowns."""

    metadata = {"render_modes": []}

    def __init__(
        self,
        scenario: Scenario | None = None,
        drive_config: DriveConfig | None = None,
        reward_config: DriveRewardConfig | None = None,
        max_episode_steps: int = 2400,
        opponents: Mapping[str, BaseAgent] | None = None,
    ) -> None:
        super().__init__()
        self.scenario = scenario if scenario is not None else RUSHED_SCENARIO
        self.drive_config = drive_config if drive_config is not None else DriveConfig()
        self.reward_config = reward_config if reward_config is not None else DriveRewardConfig()
        self.max_episode_steps = int(max_episode_steps)
        self.opponents = dict(opponents or {})
        self.action_space = spaces.Box(-1.0, 1.0, shape=(ACTION_SIZE,), dtype=np.float32)
        self.observation_space = spaces.Box(
            -1.0,
            1.0,
            shape=(DRIVE_OBSERVATION_SIZE,),
            dtype=np.float32,
        )
        self.drive = Drive(self.scenario.config, self.drive_config)
        self.play = self.drive.start_play()
        self.play_history: list[dict[str, Any]] = []
        self._agents: dict[str, BaseAgent] = {}
        self._build_agents()
        self._steps = 0
        self._episode_seed = self.scenario.seed
        self.attempt_scale = 1.0

    def set_attempt_scale(self, scale: float) -> None:
        """Scale the attempt shaping, used to anneal it during training."""
        self.attempt_scale = max(0.0, float(scale))

    def reset(
        self,
        *,
        seed: int | None = None,
        options: dict[str, Any] | None = None,
    ) -> tuple[np.ndarray, dict[str, Any]]:
        """Start a new drive; pass ``seed`` for a reproducible episode."""
        super().reset(seed=seed)
        if seed is not None:
            self._episode_seed = int(seed)
        else:
            self._episode_seed = int(self.np_random.integers(0, 2**31 - 1))
        self.drive = Drive(self.scenario.config, self.drive_config, seed=self._episode_seed)
        self.play_history = []
        self._steps = 0
        self._start_play()
        return self._observation(), self._info()

    def step(
        self,
        action: np.ndarray,
    ) -> tuple[np.ndarray, float, bool, bool, dict[str, Any]]:
        """Apply the QB action and advance the play, starting the next down if needed."""
        engine_actions: dict[str, Move | Throw] = {}
        for player_id, agent in self._agents.items():
            agent_action = agent.decide(self.play.state)
            if agent_action is not None:
                engine_actions[player_id] = agent_action
        decoded = decode_action(action, self.play.state.ball.status)
        attempted = isinstance(decoded.get("qb"), Throw)
        elapsed_before = self.play.state.elapsed_seconds
        engine_actions.update(decoded)

        terminated = False
        truncated = False
        reward = self.reward_config.play.time_penalty
        if attempted and elapsed_before >= self.reward_config.play.attempt_min_seconds - 1e-9:
            reward += self.reward_config.play.attempt * self.attempt_scale

        state = self.play.step(engine_actions)
        self._steps += 1
        if state.outcome is not PlayOutcome.IN_PROGRESS:
            down_before = self.drive.state.down
            distance_before = self.drive.state.distance
            los_before = self.drive.state.los
            result = self.drive.advance(self.play)
            reward += drive_result_reward(result, self.reward_config)
            self.play_history.append(
                {
                    "down": down_before,
                    "distance": distance_before,
                    "los_before": los_before,
                    "outcome": result.outcome.value,
                    "spot": result.spot,
                    "yards": result.yards,
                    "first_down": result.first_down,
                    "touchdown": result.touchdown,
                    "throw_distance": result.throw_distance,
                    "sack": result.outcome is PlayOutcome.SACK,
                    "coverage_sack": result.coverage_sack,
                }
            )
            terminated = self.drive.state.terminated
            if not terminated:
                self._start_play()
        if not terminated and self._steps >= self.max_episode_steps:
            truncated = True
        return self._observation(), float(reward), terminated, truncated, self._info()

    def render(self) -> None:
        """No rendering for the drive training environment."""
        return None

    def _start_play(self) -> None:
        self.play = self.drive.start_play()
        self._build_agents()

    def _build_agents(self) -> None:
        self._agents = build_rule_agents(
            self.play,
            self.drive.routes,
            self.scenario.config,
            include_quarterback=False,
        )
        self._agents.update(self.opponents)

    def _observation(self) -> np.ndarray:
        return drive_observation_from_state(
            state=self.play.state,
            routes=self.drive.routes,
            drive_state=self.drive.state,
            config=self.scenario.config,
            goal_line=self.drive_config.goal_line,
        )

    def _info(self) -> dict[str, Any]:
        return {
            "episode_seed": self._episode_seed,
            "scenario": self.scenario.name,
            "down": self.drive.state.down,
            "distance": self.drive.state.distance,
            "los": self.drive.state.los,
            "play_number": self.drive.state.play_number,
            "points": self.drive.state.points,
            "first_downs": self.drive.state.first_downs,
            "outcome": self.play.state.outcome.value,
            "drive_end_reason": self.drive.state.end_reason,
            "throw_time": self.play.state.throw_time,
            "steps": self._steps,
            "plays": list(self.play_history),
        }
```

```python
class BehaviorCloningNetwork(nn.Module):
    """MLP matching the structure of Stable-Baselines3's policy network."""

    def __init__(
        self,
        input_size: int,
        hidden: tuple[int, ...] = (64, 64),
        action_size: int = ACTION_SIZE,
    ) -> None:
        super().__init__()
        layers: list[nn.Module] = []
        size = input_size
        for hidden_size in hidden:
            layers.extend((nn.Linear(size, hidden_size), nn.Tanh()))
            size = hidden_size
        self.policy_net = nn.Sequential(*layers)
        self.action_net = nn.Linear(size, action_size)

    def forward(self, observations: torch.Tensor) -> torch.Tensor:
        return self.action_net(self.policy_net(observations))
```

```python
def collect_drive_demonstrations(
    scenario: Scenario | None = None,
    episodes: int = 200,
    seed_start: int = 0,
    drive_config: DriveConfig | None = None,
) -> DemonstrationSet:
    """Run the rule QB inside a drive and record ``(observation, action)`` pairs."""
    chosen = scenario if scenario is not None else get_scenario("rushed")
    env = QBDriveEnv(chosen, drive_config=drive_config)
    quarterback = HeuristicQuarterback("qb", chosen.config)
    observations: list[np.ndarray] = []
    actions: list[np.ndarray] = []
    for episode in range(episodes):
        _observation, _ = env.reset(seed=seed_start + episode)
        terminated = truncated = False
        while not (terminated or truncated):
            engine_action = quarterback.decide(env.play.state)
            action = (
                encode_engine_action(engine_action)
                if engine_action is not None
                else _STAND_ACTION.copy()
            )
            observations.append(
                drive_observation_from_state(
                    state=env.play.state,
                    routes=env.drive.routes,
                    drive_state=env.drive.state,
                    config=chosen.config,
                    goal_line=env.drive_config.goal_line,
                )
            )
            actions.append(action)
            _observation, _, terminated, truncated, _ = env.step(action)
    env.close()
    return DemonstrationSet(
        observations=np.stack(observations).astype(np.float32),
        actions=np.stack(actions).astype(np.float32),
```

```python
"""Aggregate metrics for drive episodes over fixed seed sets."""

from collections.abc import Callable, Iterable
from dataclasses import dataclass
from math import sqrt

import numpy as np

from qb_sim.rules.drive import DriveConfig
from qb_sim.training.drive_env import QBDriveEnv

ActionFn = Callable[[QBDriveEnv, np.ndarray], np.ndarray]
THIRD_AND_LONG_YARDS = 8.0
CHECKDOWN_YARDS = 10.0


def wilson_interval(successes: int, total: int, z: float = 1.96) -> tuple[float, float]:
    """Wilson score interval for a binomial proportion."""
    if total == 0:
        return (0.0, 0.0)
    proportion = successes / total
    denominator = 1.0 + z * z / total
    center = (proportion + z * z / (2.0 * total)) / denominator
    margin = (
        z
        * sqrt(proportion * (1.0 - proportion) / total + z * z / (4.0 * total * total))
        / denominator
    )
    return (max(0.0, center - margin), min(1.0, center + margin))


@dataclass(frozen=True, slots=True)
class DriveEpisodeResult:
    """Per-drive diagnostics used by the aggregate metrics."""

    outcome: str
    points: float
    first_downs: int
    plays: tuple[dict, ...]


def run_drive_episode(
    env: QBDriveEnv,
    action_fn: ActionFn,
    seed: int,
) -> DriveEpisodeResult:
    """Run one seeded drive and collect its play history."""
    observation, _ = env.reset(seed=seed)
    terminated = truncated = False
    while not (terminated or truncated):
        action = np.asarray(action_fn(env, observation), dtype=np.float32)
        observation, _, terminated, truncated, _ = env.step(action)
    state = env.drive.state
    return DriveEpisodeResult(
        outcome=state.end_reason or "truncated",
        points=state.points,
        first_downs=state.first_downs,
        plays=tuple(env.play_history),
    )


def evaluate_drive(
    action_fn: ActionFn,
    scenario_name: str,
    seeds: Iterable[int],
    drive_config: DriveConfig | None = None,
) -> dict:
    """Evaluate a policy on fixed drive seeds and aggregate its metrics."""
    from qb_sim.rules.scenarios import get_scenario

    env = QBDriveEnv(get_scenario(scenario_name), drive_config=drive_config)
    results = [run_drive_episode(env, action_fn, seed) for seed in seeds]
    env.close()
    return summarize_drives(results)


def summarize_drives(results: list[DriveEpisodeResult]) -> dict:
    """Aggregate drive results into strategy, scoring, and efficiency metrics."""
    total = len(results)
    if total == 0:
        raise ValueError("at least one drive is required")

    def pct(count: int, denominator: int) -> float:
        return 100.0 * count / denominator if denominator else 0.0

    def ci(count: int, denominator: int) -> list[float]:
        low, high = wilson_interval(count, denominator)
        return [100.0 * low, 100.0 * high]

    touchdowns = sum(result.outcome == "touchdown" for result in results)
    interceptions = sum(result.outcome == "interception" for result in results)
    turnovers = sum(result.outcome == "turnover_on_downs" for result in results)
    plays = [play for result in results for play in result.plays]
    third_downs = [play for play in plays if play["down"] == 3]
    third_and_long = [
        play for play in third_downs if play["distance"] >= THIRD_AND_LONG_YARDS
    ]
    third_conversions = sum(
        bool(play["first_down"] or play["touchdown"]) for play in third_downs
    )
    long_conversions = sum(
        bool(play["first_down"] or play["touchdown"]) for play in third_and_long
    )
    long_throws = [
        float(play["throw_distance"])
        for play in third_and_long
        if play["throw_distance"] is not None
    ]
    checkdowns = sum(distance <= CHECKDOWN_YARDS for distance in long_throws)
    sacks = sum(play["outcome"] == "sack" for play in plays)
    coverage_sacks = sum(bool(play["coverage_sack"]) for play in plays)
    incompletes = sum(play["outcome"] == "incomplete" for play in plays)

    return {
        "episodes": float(total),
        "touchdown_pct": pct(touchdowns, total),
        "touchdown_ci": ci(touchdowns, total),
        "interception_pct": pct(interceptions, total),
        "interception_ci": ci(interceptions, total),
        "turnover_on_downs_pct": pct(turnovers, total),
        "turnover_on_downs_ci": ci(turnovers, total),
        "points_per_drive": sum(result.points for result in results) / total,
        "first_downs_per_drive": sum(result.first_downs for result in results) / total,
        "plays_per_drive": len(plays) / total,
        "third_down_conversion_pct": pct(third_conversions, len(third_downs)),
        "third_down_conversion_ci": ci(third_conversions, len(third_downs)),
        "third_and_long_conversion_pct": pct(long_conversions, len(third_and_long)),
        "third_and_long_conversion_ci": ci(long_conversions, len(third_and_long)),
        "third_and_long_throw_depth_yards": (
            sum(long_throws) / len(long_throws) if long_throws else 0.0
        ),
        "checkdown_pct": pct(checkdowns, len(long_throws)),
        "checkdown_ci": ci(checkdowns, len(long_throws)),
        "sack_pct": pct(sacks, len(plays)),
        "sack_ci": ci(sacks, len(plays)),
        "coverage_sack_pct": pct(coverage_sacks, len(plays)),
        "coverage_sack_ci": ci(coverage_sacks, len(plays)),
        "incomplete_pct": pct(incompletes, len(plays)),
        "incomplete_ci": ci(incompletes, len(plays)),
    }
```

Note: `pct`/`ci` take integer counts; `ci` on 0 denominators returns `[0.0, 0.0]` via `wilson_interval`.

```python
"""Compare the rule-based and trained quarterbacks on fixed drive seeds.

Example:

    python scripts/evaluate_drive.py \
        --ppo v11=runs/ppo_drive_qb_v11/final_model.zip \
        --episodes 100 --output runs/drive_evaluation_v11.json
"""

import argparse
import json
from pathlib import Path

from qb_sim.rules.scenarios import get_scenario
from qb_sim.training.drive_evaluation import evaluate_drive
from qb_sim.training.evaluation import PPOActionPolicy, RuleQuarterbackPolicy

DEFAULT_EPISODES = 100
DEFAULT_SEED_START = 10_000
COLUMNS = (
    "touchdown_pct",
    "points_per_drive",
    "first_downs_per_drive",
    "plays_per_drive",
    "third_down_conversion_pct",
    "third_and_long_conversion_pct",
    "third_and_long_throw_depth_yards",
    "checkdown_pct",
    "interception_pct",
    "sack_pct",
    "coverage_sack_pct",
)


def main() -> None:
    parser = argparse.ArgumentParser(description="Evaluate quarterbacks on QBDrive-v0")
    parser.add_argument("--ppo", action="append", default=[], metavar="LABEL=PATH")
    parser.add_argument("--episodes", type=int, default=DEFAULT_EPISODES)
    parser.add_argument("--seed-start", type=int, default=DEFAULT_SEED_START)
    parser.add_argument("--scenario", default="rushed")
    parser.add_argument("--output", default="runs/drive_evaluation.json")
    args = parser.parse_args()

    scenario = get_scenario(args.scenario)
    seeds = range(args.seed_start, args.seed_start + args.episodes)
    policies: list[tuple[str, object]] = [
        ("rules", RuleQuarterbackPolicy(scenario.config))
    ]
    for entry in args.ppo:
        label, path = entry.split("=", 1)
        policies.append((label, PPOActionPolicy(path)))

    results: dict[str, dict] = {}
    for label, policy in policies:
        if isinstance(policy, PPOActionPolicy) and not policy.supports_drive_layout():
            print(f"{label}: omitido (layout incompatible: {policy.observation_size})")
            continue
        results[label] = evaluate_drive(policy, args.scenario, seeds)

    print(f"\\n=== {args.scenario} ({args.episodes} drives, semillas {args.seed_start}+) ===")
    header = f"{'policy':<12}" + "".join(f"{column[:11]:>12}" for column in COLUMNS)
    print(header)
    for name, summary in results.items():
        row = f"{name:<12}" + "".join(f"{summary[column]:>12.2f}" for column in COLUMNS)
        print(row)

    output_path = Path(args.output)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(json.dumps(results, indent=2) + "\\n", encoding="utf-8")
    print(f"\\nsaved {output_path}")


if __name__ == "__main__":
    main()
```
